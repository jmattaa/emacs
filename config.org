* Project.el
#+begin_src elisp
  (require 'project)

  (defun jmattaa/scan-projects ()
    "Recursively discover Git repositories under ~/dev."
    (interactive)
    (let ((root (expand-file-name "~/dev/")))
      (dolist (dir (directory-files-recursively
  		  root
  		  "\\`\\.git\\'"
  		  t
  		  (lambda (dir)
  		    (not (member (file-name-nondirectory dir)
  				 '(".git" "node_modules" "vendor"))))))
        (let ((project-root (file-name-directory dir)))
  	(project-remember-project
  	 (project-current nil project-root))))
      (message "Project scan complete")))
  (add-hook 'emacs-startup-hook #'jmattaa/scan-projects)
#+end_src
